function onCreated() {
  this.user.clearVars(); // don't touch this
  
  // users must use the account name since folder rights are derived
  // from that user's RC rights
  this.user.moondeath = true;
  this.user.moondeath.password = md5("test"); // you can use a pre-MD5'd value here
  this.user.moondeath.allowWrite = true;
  
  this.logFile = "ftplog.txt"; // don't prefix with logs/; logging is mandatory
  
  this.allowWrite = true; // allow writing (must also be true for
                          // each user that can write), this takes
                          // precedent over user-specific rights, so
                          // false won't allow any users to write
  
  // don't touch below here
  
  // get server IP
  temp.req = requestURL("https://4.ident.me");
  
  waitfor(req, "onReceiveData", 30);
  //echo(req.data);
  this.ip = "127.0.0.1";//req.data[0];
  this.port = 4201;
  //echo("IP:" SPC this.ip);
  // destroy if existing
  if (isObject("FTPListenSocket")) {
    FTPListenSocket.destroy();
  }
  
  // create the socket
  this.listenSocket = new TSocket("FTPListenSocket");
  this.listenSocket.optionDB = this;
  this.listenSocket.join("ftp_listensocket");
}

function onInitialized() {
  this.trigger("created");
}

public function onShowListenInfo() {
  echo("FTP: Listening on " @ this.ip @ ":" @ this.listenSocket.port);
}

public function isValidFolderRight(acc, mode, file) {

  //return;     
  // TServerPlayer.hasRight(mode, file)
  temp.pl = findPlayer(acc);
  temp.isOffline = (pl == null);
  temp.hasRight = false;
  
  if (isOffline) {
    pl = new TServerPlayer(@ acc);
    hasRight = pl.hasRight(mode, file);
    pl.destroy();
  } else {
    hasRight = pl.hasRight(mode, file);
  }
  echo(file SPC hasRight);
  return hasRight;
}
