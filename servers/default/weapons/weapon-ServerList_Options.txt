function onCreated() {
  this.keyicons = {"up","left","down","right",
    "shoot","sword","grab","map","chat","items","pause"};
  this.keynames = {"Up","Left","Down","Right",
    "Shoot/Weapons","Sword","Grab/Lift","Map",
    "Chat","Inventory","Pause"};

  // Check setting the gui style
  temp.updateoptions = false;
  if (graalversion >= 6.030 && $pref::Video::defaultguistyle=="" &&
      !client.defaultstyleset) {
    client.defaultstyleset = true;
    $pref::Video::defaultguistyle = "vplusblue.wba";
    temp.updateoptions = true;
  }
  if (graalversion >= 6.035 && $pref::Video::externalguistyle=="" &&
      !client.externalstyleset) {
    client.externalstyleset = true;
    $pref::Video::externalguistyle = $pref::Video::defaultguistyle;
    temp.updateoptions = true;
  }
  if (temp.updateoptions)
    Adventure_saveGraalOptions();
  showOptions();
}

public function getGUIStyle() {
  if (isObject("-Serverlist"))
    return ("-Serverlist").getExternalGUIStyle();
  else if (graalversion >= 6.035)
    return $pref::Video::externalguistyle;
  else
    return $pref::Video::defaultguistyle;
}

function onKeypressed(keycode) {
  if (graalversion >= 5.306)
    return;
  if (keycode==0x72) { // F3
    if (isObject("OptDialogWindow2D") &&
        OptDialogWindow2D.visible)
      OptDialogWindow2D.visible = false;
    else
      showOptions();
  }
}

function onControlKeyDown(keycode,keytext,scancode,window) {
  if (keycode==VK_F3 ||
      (window=="OptDialogWindow2D" && keycode==VK_ESCAPE)) {
    if (OptDialogWindow2D.visible)
      OptDialogWindow2D.visible = false;
    else
      showOptions();
  }
}

public function showOptions() {
  cursoron();
  addGuiControls();
  addSubControls();
  addInputCtrl();
  OptDialogWindow2D.showtop();
}

function onPlayerLanguageChanges() {
  if (isObject("OptDialogWindow2D"))
    addSubControls();
}

function onDefaultStyleChanges() {
  if (graalversion < 6.035 && isObject("OptDialogWindow2D"))
    addSubControls();
}

function onExternalStyleChanges() {
  if (graalversion >= 6.035 && isObject("OptDialogWindow2D"))
    addSubControls();
}

function OptDialogWindow2D.onHide() {
  // Apply all options
  
  // GUI style
  temp.newstyle = OptDefaultStyleMenu2D.text;
  $pref::Video::defaultguistyle = (temp.newstyle!="Default" && 
    temp.newstyle!=_("Default")? temp.newstyle @ ".wba" : "");
  if (graalversion >= 6.035) {
    temp.newstyle = OptExternalStyleMenu2D.text;
    $pref::Video::externalguistyle = (temp.newstyle!="Default" && 
      temp.newstyle!=_("Default")? temp.newstyle @ ".wba" : "");
  }

  // General options
  $pref::Graal::automapping = OptAutomappingToggle2D.checked;
  $pref::Graal::notoalls = !OptToallsToggle2D.checked;
  if (isObject("OptFilenameScanToggle2D"))
    $pref::Graal::filescan = OptFilenameScanToggle2D.checked;
  if (isObject("OptFilenameCacheToggle2D"))
    $pref::Graal::cachefilenames = OptFilenameCacheToggle2D.checked;
  $pref::Graal::limitnicknames = OptLimitnicknamesToggle2D.checked;
  $pref::Graal::nicknamelimit = OptNicknamelengthField2D.text;
  $pref::Graal::defaultfontsize = OptFontsizeField2D.text;
  $pref::Graal::defaultfontname = OptDefaultFontMenu2D.text;

  $pref::Audio::reversestereo = OptReversestereoToggle2D.checked;

  // Video recording
  $pref::Capture::size = OptVideoSizeField2D.options[OptVideoSizeField2D.getSelectedRow()];
  $pref::Capture::area = OptCaptureAreaField2D.options[OptCaptureAreaField2D.getSelectedRow()];
  $pref::Capture::fps = OptFrameRateField2D.text;
  $pref::Capture::disabledaynight = OptDisableCaptureDayNightToggle2D.checked;
  $pref::Capture::hideinterface = OptHideInterfaceToggle2D.checked;
  $pref::Capture::showmouse = OptShowCaptureMouseToggle2D.checked;
  $pref::Capture::recordaudio = OptRecordAudioToggle2D.checked;
  Adventure_setAllowRecordByScript(OptCaptureByScriptToggle2D.checked);
  $pref::Capture::format = OptRecordFormatField2D.formats[OptRecordFormatField2D.getSelectedRow()];
  $pref::Capture::videobitrate = OptVideoBitRateField2D.value * 1000;
  $pref::Capture::audiobitrate = OptAudioBitRateField2D.value * 1000;

  // Switch between OpenGL and DirectX
  $pref::Video::fullscreenmode = OptGraphicsResolutionMenu2D.text;
  if (isObject("OptOpenGLToggle2D") && OptOpenGLToggle2D.visible) {
    temp.useopengl = OptOpenGLToggle2D.checked;
    if ($pref::Video::useOpenGL != temp.useopengl) {
      // Destroy some windows because the window recreation for OpenGL
      // is messing them up
      if (isObject("PlayerList_Window")) {
        PlayerList_Window.destroy();
        PlayerList_Window = NULL;
      }
      OptDialogWindow2D.destroy();
      OptDialogWindow2D = NULL;
    }
    $pref::Video::useOpenGL = temp.useopengl;
  }

  // Save controls and options
  Adventure_saveGraalOptions();
}

function OptGraphicsDriverMenu2D_init(device) {
  if (!isObject("OptGraphicsResolutionMenu2D"))
    return;

  // Attempt to keep the same resolution settings:
  if (OptGraphicsResolutionMenu2D.rows.size() > 0)
    temp.prevRes = OptGraphicsResolutionMenu2D.text;
  else
    temp.prevRes = $pref::Video::fullScreenMode;

  // Fill the resolution lists:
  OptGraphicsResolutionMenu2D_init(device);

  // Try to select the previous settings:
  temp.selId = OptGraphicsResolutionMenu2D.findTextId(temp.prevRes);
  if (temp.selId < 0)
    temp.selId = 0;
  OptGraphicsResolutionMenu2D.setSelectedById(temp.selId);
}

function OptGraphicsResolutionMenu2D_init(device) {
  OptGraphicsResolutionMenu2D.clearRows();
  reslist = getResolutionList( device );

  temp.count = 0;
  for (temp.i = 0; temp.i < reslist.size(); temp.i++ ) {
    //Note: for DirectX9 it's also specifying the refresh rate,
    // but it's using the default rate anyway, so ignore it
    temp.resarr = reslist[temp.i].tokenize("x");
    if (temp.resarr.size() < 2)
      continue;
    res = temp.resarr[0] @ "x" @ temp.resarr[1];
    // Only add to list if it isn't there already:
    if (OptGraphicsResolutionMenu2D.findText(res ) >= 0)
      continue;
      
    OptGraphicsResolutionMenu2D.addRow(temp.count, res);
    temp.count++;
  }
}

function OptRemapList2D_fillList() {
  if (!scriptedcontrols)
    return;

  with (OptRemapList2D) {
    clearRows();
    for (temp.i=0; temp.i<thiso.keyicons.size(); temp.i++) {
      with (addRow(temp.i, buildFullMapString(temp.i, -1, ""))) {
        icon.clearall();
        icon.drawimage(0,0,"controls_" @ thiso.keyicons[temp.i] @ ".png");
      }
    }
  }
}

function buildFullMapString(action, replaceindex, replacestr) {
  temp.txt = thiso.keynames[action];
  for (temp.i=0; temp.i<3; temp.i++) {
    temp.txt @= "\t";
    if (temp.i==replaceindex)
      temp.txt @= replacestr;
    else
      temp.txt @= Adventure_getControlBinding(action, temp.i).keytext.substring(0,9);
  }
  return temp.txt;
}

function OptRemapList2D_doRemap(action, slot) {
//  selid = OptRemapList2D.getSelectedId();

  this.remapmode = "2D";
  this.remapindex = action;
  this.remapslot = slot;

  OptRemapList2D.setRowById(action, buildFullMapString(action, slot, "Type a key"));

  OptDialogWindow2D.addcontrol(RemapDlg);
  pushDialog(RemapDlg);
}

function RemapDlg.onKeyDown(keycode,keychar,scancode) {
//  echo("key down: " @ params);
  finishRemapDialog(keycode, keychar);
}

function RemapDlg.onMouseDown() {
  finishRemapDialog(-1, "");
}

function RemapDlg.onRightMouseDown() {
  finishRemapDialog(-1, "");
}

function finishRemapDialog(keycode, keychar) {
  if ((keycode & 0xff) in {0x10, 0x11, 0x12, 0xA0, 0xA1, 0xA2, 0xA3, 0xA4, 0xA5})
    return;

  if (keycode>=0)
    Adventure_setControlBinding(this.remapindex, this.remapslot, 
      (keycode==0x2E? -1 : keycode & 0xff)); // 0x2E = VK_ESCAPE
      
  OptRemapList2D_fillList();

  popDialog();
}

function OptRemapList2D.onMouseDown(mousemod, mx, my, clickcount) {
  temp.pos = OptRemapList2D.globaltolocalcoord({mx,my});
  temp.row = OptRemapList2D.getRowAtPoint(mx, my);
  temp.col = -1;
  for (temp.colx: OptRemapList2D.columns) {
    if (temp.pos[0]<temp.colx)
      break;
    temp.col++;
  }

    OptRemapList2D_doRemap(temp.row, temp.col-1);
}
